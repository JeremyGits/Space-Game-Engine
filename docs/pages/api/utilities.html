MathUtils

Collection of mathematical utility functions:

src/utils/math/MathUtils.ts
class MathUtils {
  // Constants
  static readonly PI = Math.PI;
  static readonly TWO_PI = Math.PI * 2;
  static readonly HALF_PI = Math.PI / 2;
  static readonly DEG_TO_RAD = Math.PI / 180;
  static readonly RAD_TO_DEG = 180 / Math.PI;
  static readonly EPSILON = 0.000001;
  
  // Clamping
  static clamp(value: number, min: number, max: number): number {
    return Math.max(min, Math.min(max, value));
  }
  
  static clamp01(value: number): number {
    return MathUtils.clamp(value, 0, 1);
  }
  
  // Interpolation
  static lerp(a: number, b: number, t: number): number {
    return a + (b - a) * t;
  }
  
  static inverseLerp(a: number, b: number, value: number): number {
    return (value - a) / (b - a);
  }
  
  static smoothstep(a: number, b: number, t: number): number {
    t = MathUtils.clamp01((t - a) / (b - a));
    return t * t * (3 - 2 * t);
  }
  
  static smootherstep(a: number, b: number, t: number): number {
    t = MathUtils.clamp01((t - a) / (b - a));
    return t * t * t * (t * (t * 6 - 15) + 10);
  }
  
  // Angle utilities
  static degToRad(degrees: number): number {
    return degrees * MathUtils.DEG_TO_RAD;
  }
  
  static radToDeg(radians: number): number {
    return radians * MathUtils.RAD_TO_DEG;
  }
  
  static normalizeAngle(angle: number): number {
    while (angle > Math.PI) angle -= MathUtils.TWO_PI;
    while (angle < -Math.PI) angle += MathUtils.TWO_PI;
    return angle;
  }
  
  static shortestAngle(from: number, to: number): number {
    const diff = MathUtils.normalizeAngle(to - from);
    return diff;
  }
  
  // Comparison
  static approximately(a: number, b: number, epsilon = MathUtils.EPSILON): boolean {
    return Math.abs(a - b) < epsilon;
  }
  
  static sign(value: number): number {
    return value >= 0 ? 1 : -1;
  }
  
  // Mapping
  static map(value: number, inMin: number, inMax: number, outMin: number, outMax: number): number {
    return outMin + (outMax - outMin) * ((value - inMin) / (inMax - inMin));
  }
  
  // Random
  static random(min: number = 0, max: number = 1): number {
    return min + Math.random() * (max - min);
  }
  
  static randomInt(min: number, max: number): number {
    return Math.floor(MathUtils.random(min, max + 1));
  }
  
  static randomBool(): boolean {
    return Math.random() < 0.5;
  }
}

Interpolation

src/utils/math/Interpolation.ts
class Interpolation {
  // Linear interpolation
  static linear(a: number, b: number, t: number): number {
    return a + (b - a) * t;
  }
  
  // Quadratic easing
  static quadraticIn(a: number, b: number, t: number): number {
    return a + (b - a) * t * t;
  }
  
  static quadraticOut(a: number, b: number, t: number): number {
    return a + (b - a) * t * (2 - t);
  }
  
  static quadraticInOut(a: number, b: number, t: number): number {
    t *= 2;
    if (t < 1) return a + (b - a) * 0.5 * t * t;
    t--;
    return a + (b - a) * (-0.5 * (t * (t - 2) - 1));
  }
  
  // Cubic easing
  static cubicIn(a: number, b: number, t: number): number {
    return a + (b - a) * t * t * t;
  }
  
  static cubicOut(a: number, b: number, t: number): number {
    t--;
    return a + (b - a) * (t * t * t + 1);
  }
  
  // Exponential easing
  static exponentialIn(a: number, b: number, t: number): number {
    return a + (b - a) * Math.pow(2, 10 * (t - 1));
  }
  
  static exponentialOut(a: number, b: number, t: number): number {
    return a + (b - a) * (-Math.pow(2, -10 * t) + 1);
  }
  
  // Elastic easing
  static elasticOut(a: number, b: number, t: number): number {
    const p = 0.3;
    return a + (b - a) * (Math.pow(2, -10 * t) * Math.sin((t - p / 4) * (2 * Math.PI) / p) + 1);
  }
  
  // Bounce easing
  static bounceOut(a: number, b: number, t: number): number {
    if (t < 1 / 2.75) {
      return a + (b - a) * (7.5625 * t * t);
    } else if (t < 2 / 2.75) {
      t -= 1.5 / 2.75;
      return a + (b - a) * (7.5625 * t * t + 0.75);
    } else if (t < 2.5 / 2.75) {
      t -= 2.25 / 2.75;
      return a + (b - a) * (7.5625 * t * t + 0.9375);
    } else {
      t -= 2.625 / 2.75;
      return a + (b - a) * (7.5625 * t * t + 0.984375);
    }
  }
}

Easing Functions

class Easing {
  // Sine easing
  static sineIn(t: number): number {
    return 1 - Math.cos(t * Math.PI / 2);
  }
  
  static sineOut(t: number): number {
    return Math.sin(t * Math.PI / 2);
  }
  
  static sineInOut(t: number): number {
    return -(Math.cos(Math.PI * t) - 1) / 2;
  }
  
  // Circular easing
  static circIn(t: number): number {
    return 1 - Math.sqrt(1 - t * t);
  }
  
  static circOut(t: number): number {
    t--;
    return Math.sqrt(1 - t * t);
  }
  
  // Back easing (overshoots)
  static backIn(t: number): number {
    const c = 1.70158;
    return t * t * ((c + 1) * t - c);
  }
  
  static backOut(t: number): number {
    const c = 1.70158;
    t--;
    return t * t * ((c + 1) * t + c) + 1;
  }
}

AABB (Axis-Aligned Bounding Box)

src/utils/geometry/AABB.ts
class AABB {
  min: Vector3;
  max: Vector3;
  
  constructor(min?: Vector3, max?: Vector3) {
    this.min = min || new Vector3();
    this.max = max || new Vector3();
  }
  
  // Creation
  static fromPoints(points: Vector3[]): AABB {
    const min = new Vector3(Infinity, Infinity, Infinity);
    const max = new Vector3(-Infinity, -Infinity, -Infinity);
    
    for (const point of points) {
      min.x = Math.min(min.x, point.x);
      min.y = Math.min(min.y, point.y);
      min.z = Math.min(min.z, point.z);
      max.x = Math.max(max.x, point.x);
      max.y = Math.max(max.y, point.y);
      max.z = Math.max(max.z, point.z);
    }
    
    return new AABB(min, max);
  }
  
  // Properties
  getCenter(): Vector3 {
    return this.min.clone().add(this.max).multiply(0.5);
  }
  
  getSize(): Vector3 {
    return this.max.clone().subtract(this.min);
  }
  
  getVolume(): number {
    const size = this.getSize();
    return size.x * size.y * size.z;
  }
  
  // Tests
  containsPoint(point: Vector3): boolean {
    return point.x >= this.min.x && point.x <= this.max.x &&
           point.y >= this.min.y && point.y <= this.max.y &&
           point.z >= this.min.z && point.z <= this.max.z;
  }
  
  intersectsAABB(other: AABB): boolean {
    return this.min.x <= other.max.x && this.max.x >= other.min.x &&
           this.min.y <= other.max.y && this.max.y >= other.min.y &&
           this.min.z <= other.max.z && this.max.z >= other.min.z;
  }
  
  intersectsSphere(sphere: Sphere): boolean {
    const closestPoint = this.closestPoint(sphere.center);
    return closestPoint.distanceTo(sphere.center) <= sphere.radius;
  }
  
  closestPoint(point: Vector3): Vector3 {
    return new Vector3(
      MathUtils.clamp(point.x, this.min.x, this.max.x),
      MathUtils.clamp(point.y, this.min.y, this.max.y),
      MathUtils.clamp(point.z, this.min.z, this.max.z)
    );
  }
  
  // Operations
  expand(amount: number): void {
    this.min.subtract(new Vector3(amount, amount, amount));
    this.max.add(new Vector3(amount, amount, amount));
  }
  
  merge(other: AABB): void {
    this.min.x = Math.min(this.min.x, other.min.x);
    this.min.y = Math.min(this.min.y, other.min.y);
    this.min.z = Math.min(this.min.z, other.min.z);
    this.max.x = Math.max(this.max.x, other.max.x);
    this.max.y = Math.max(this.max.y, other.max.y);
    this.max.z = Math.max(this.max.z, other.max.z);
  }
}

Sphere

src/utils/geometry/Sphere.ts
class Sphere {
  center: Vector3;
  radius: number;
  
  constructor(center?: Vector3, radius?: number) {
    this.center = center || new Vector3();
    this.radius = radius || 1;
  }
  
  // Creation
  static fromPoints(points: Vector3[]): Sphere {
    // Calculate bounding sphere
    const center = new Vector3();
    for (const point of points) {
      center.add(point);
    }
    center.divide(points.length);
    
    let radius = 0;
    for (const point of points) {
      const dist = center.distanceTo(point);
      radius = Math.max(radius, dist);
    }
    
    return new Sphere(center, radius);
  }
  
  // Tests
  containsPoint(point: Vector3): boolean {
    return this.center.distanceTo(point) <= this.radius;
  }
  
  intersectsSphere(other: Sphere): boolean {
    const distance = this.center.distanceTo(other.center);
    return distance <= (this.radius + other.radius);
  }
  
  intersectsAABB(aabb: AABB): boolean {
    const closestPoint = aabb.closestPoint(this.center);
    return this.containsPoint(closestPoint);
  }
  
  // Operations
  expand(amount: number): void {
    this.radius += amount;
  }
  
  merge(other: Sphere): void {
    const direction = other.center.clone().subtract(this.center);
    const distance = direction.length();
    
    if (distance + other.radius <= this.radius) return;
    if (distance + this.radius <= other.radius) {
      this.center.copy(other.center);
      this.radius = other.radius;
      return;
    }
    
    const newRadius = (this.radius + distance + other.radius) / 2;
    this.center.add(direction.normalize().multiply(newRadius - this.radius));
    this.radius = newRadius;
  }
}

class Clock {
  private startTime: number;
  private lastTime: number;
  private elapsedTime: number = 0;
  private running: boolean = false;
  
  start(): void {
    this.startTime = performance.now();
    this.lastTime = this.startTime;
    this.running = true;
  }
  
  stop(): void {
    this.getElapsedTime();
    this.running = false;
  }
  
  getElapsedTime(): number {
    if (this.running) {
      const now = performance.now();
      this.elapsedTime = (now - this.startTime) / 1000;
    }
    return this.elapsedTime;
  }
  
  getDelta(): number {
    if (!this.running) return 0;
    
    const now = performance.now();
    const delta = (now - this.lastTime) / 1000;
    this.lastTime = now;
    return delta;
  }
  
  reset(): void {
    this.startTime = performance.now();
    this.lastTime = this.startTime;
    this.elapsedTime = 0;
  }
}

Timer

Countdown timer with callbacks:

src/utils/time/Timer.ts
class Timer {
  private duration: number;
  private elapsed: number = 0;
  private running: boolean = false;
  private onComplete?: () => void;
  private onTick?: (remaining: number) => void;
  
  constructor(duration: number) {
    this.duration = duration;
  }
  
  start(onComplete?: () => void, onTick?: (remaining: number) => void): void {
    this.running = true;
    this.elapsed = 0;
    this.onComplete = onComplete;
    this.onTick = onTick;
  }
  
  update(deltaTime: number): void {
    if (!this.running) return;
    
    this.elapsed += deltaTime;
    
    const remaining = this.duration - this.elapsed;
    
    if (this.onTick) {
      this.onTick(Math.max(0, remaining));
    }
    
    if (this.elapsed >= this.duration) {
      this.running = false;
      if (this.onComplete) {
        this.onComplete();
      }
    }
  }
  
  getRemaining(): number {
    return Math.max(0, this.duration - this.elapsed);
  }
  
  getProgress(): number {
    return MathUtils.clamp01(this.elapsed / this.duration);
  }
  
  reset(): void {
    this.elapsed = 0;
    this.running = false;
  }
}

// Usage
const timer = new Timer(5.0); // 5 second timer

timer.start(
  () => console.log('Timer complete!'),
  (remaining) => console.log(`${remaining.toFixed(1)}s remaining`)
);

// In game loop
timer.update(deltaTime);

Stopwatch

Measures elapsed time with lap support:

src/utils/time/Stopwatch.ts
class Stopwatch {
  private startTime: number = 0;
  private elapsed: number = 0;
  private running: boolean = false;
  private laps: number[] = [];
  
  start(): void {
    if (!this.running) {
      this.startTime = performance.now() - this.elapsed * 1000;
      this.running = true;
    }
  }
  
  stop(): void {
    if (this.running) {
      this.elapsed = (performance.now() - this.startTime) / 1000;
      this.running = false;
    }
  }
  
  reset(): void {
    this.startTime = performance.now();
    this.elapsed = 0;
    this.laps = [];
  }
  
  lap(): number {
    const lapTime = this.getElapsed();
    this.laps.push(lapTime);
    return lapTime;
  }
  
  getElapsed(): number {
    if (this.running) {
      return (performance.now() - this.startTime) / 1000;
    }
    return this.elapsed;
  }
  
  getLaps(): number[] {
    return [...this.laps];
  }
  
  getBestLap(): number {
    return Math.min(...this.laps);
  }
}

// Usage
const stopwatch = new Stopwatch();

stopwatch.start();
// ... do something ...
const lap1 = stopwatch.lap();
// ... do something else ...
const lap2 = stopwatch.lap();
stopwatch.stop();

console.log(`Total: ${stopwatch.getElapsed()}s`);
console.log(`Best lap: ${stopwatch.getBestLap()}s`);

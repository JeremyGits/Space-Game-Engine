Player Component

Player entity with first-person controller:

function Player({ inputManager }: { inputManager: InputManager }) {
  const { camera } = useThree();
  const controllerRef = useRef(new FirstPersonController());
  
  useFrame((_, delta) => {
    // Get input
    const input = {
      moveForward: inputManager.getAxis('moveForward'),
      moveRight: inputManager.getAxis('moveRight'),
      lookDeltaX: mouseDelta.x,
      lookDeltaY: mouseDelta.y,
      sprint: inputManager.getAction('sprint') > 0.5,
      crouch: false,
      jump: false
    };
    
    // Update controller
    controllerRef.current.update(input, delta);
    
    // Apply to camera
    camera.position.copy(controllerRef.current.getCameraPosition());
    camera.quaternion.copy(controllerRef.current.getCameraQuaternion());
  });
  
  return null;
}

Terrain Component

Procedural terrain with physics:

function Terrain() {
  const terrainRef = useRef(null);
  
  useEffect(() => {
    if (!terrainRef.current) return;
    
    const geometry = terrainRef.current.geometry as THREE.PlaneGeometry;
    const positions = geometry.attributes.position;
    
    // Generate terrain heights
    for (let i = 0; i < positions.count; i++) {
      const x = positions.getX(i);
      const y = positions.getY(i);
      
      const height = 
        Math.sin(x * 0.1) * Math.cos(y * 0.1) * 2 +
        Math.sin(x * 0.05) * Math.cos(y * 0.05) * 3;
      
      positions.setZ(i, height);
    }
    
    positions.needsUpdate = true;
    geometry.computeVertexNormals();
  }, []);

  );
}

GrassField Component

GPU-instanced grass field:

function GrassField() {
  const grassRef = useRef(null);
  const { camera } = useThree();
  
  useEffect(() => {
    grassRef.current = new GrassRenderer({
      density: 8,
      areaSize: 70,
      bladeHeight: 0.15,
      bladeWidth: 0.02,
      colorVariation: 0.3,
      windStrength: 0.3,
      lodDistance: 40
    });
    
    grassRef.current.generateGrassField((x, z) => {
      return Math.sin(x * 0.1) * Math.cos(z * 0.1) * 2;
    });
    
    return () => grassRef.current?.dispose();
  }, []);
  
  useFrame((_, delta) => {
    if (grassRef.current) {
      grassRef.current.update(delta);
      grassRef.current.updateLOD(camera.position);
    }
  });
  
  return grassRef.current ?  : null;
}

Helper Components

PositionTracker

Tracks camera position and performance stats:

function PositionTracker({ 
  onPositionUpdate, 
  onStatsUpdate 
}: PositionTrackerProps) {
  const { camera, gl } = useThree();
  const frameCount = useRef(0);
  const lastTime = useRef(performance.now());
  
  useFrame(() => {
    // Update position
    onPositionUpdate({
      x: camera.position.x,
      y: camera.position.y,
      z: camera.position.z
    });
    
    // Update stats every 30 frames
    frameCount.current++;
    if (frameCount.current % 30 === 0) {
      const now = performance.now();
      const fps = Math.round(1000 / ((now - lastTime.current) / 30));
      lastTime.current = now;
      
      onStatsUpdate({
        fps,
        triangles: gl.info.render.triangles
      });
    }
  });
  
  return null;
}

interface PositionTrackerProps {
  onPositionUpdate: (pos: { x: number; y: number; z: number }) => void;
  onStatsUpdate: (stats: { fps: number; triangles: number }) => void;
}

Using useThree Hook

Access Three.js scene, camera, and renderer:

import { useThree } from '@react-three/fiber';

function CameraController() {
  const { camera, scene, gl } = useThree();
  
  useEffect(() => {
    // Configure camera
    camera.fov = 75;
    camera.updateProjectionMatrix();
    
    // Access renderer
    gl.shadowMap.enabled = true;
    gl.shadowMap.type = THREE.PCFSoftShadowMap;
  }, [camera, gl]);
  
  return null;
}

Best Practices

Performance

  • Use refs for Three.js objects that update frequently
  • Avoid creating new objects in useFrame
  • Use useMemo for expensive calculations
  • Implement LOD for complex scenes
  • Use GPU instancing for many similar objects

React Three Fiber

  • Use lowercase for Three.js primitives (, )
  • Use for existing Three.js objects
  • Props map directly to Three.js properties
  • Use attach prop for non-standard attachments

Component Organization

  • Keep game components in src/components/game/
  • Keep UI components in src/components/ui/
  • Extract reusable logic into custom hooks
  • Use TypeScript interfaces for props

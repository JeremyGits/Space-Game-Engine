⚙️ Core Systems

Deep dive into the engine's core systems and implementation

GameEngine

The GameEngine class is the main orchestrator that initializes and manages all engine systems.

src/engine/core/GameEngine.ts
class GameEngine {
  private modules: Map;
  private systems: SystemManager;
  private loop: GameLoop;
  private scene: SceneManager;
  private isRunning: boolean = false;
  
  constructor(config: EngineConfig) {
    this.modules = new Map();
    this.systems = new SystemManager();
    this.loop = new GameLoop();
    this.scene = new SceneManager();
  }
  
  async initialize(): Promise {
    // Initialize all modules
    for (const module of this.modules.values()) {
      await module.initialize();
    }
    
    // Start game loop
    this.loop.start(this.tick.bind(this));
    this.isRunning = true;
  }
  
  private tick(deltaTime: number): void {
    // Update all systems
    this.systems.update(deltaTime);
    
    // Update scene
    this.scene.update(deltaTime);
  }
  
  shutdown(): void {
    this.loop.stop();
    this.isRunning = false;
    
    // Cleanup modules
    for (const module of this.modules.values()) {
      module.shutdown();
    }
  }
}

Game Loop

The game loop uses a fixed timestep for physics and variable timestep for rendering, ensuring consistent simulation regardless of framerate.

Fixed Timestep Implementation

src/engine/core/GameLoop.ts
class GameLoop {
  private readonly FIXED_TIMESTEP = 1 / 60; // 60 Hz
  private readonly MAX_FRAME_TIME = 0.25;   // Cap at 250ms
  
  private accumulator: number = 0;
  private lastTime: number = 0;
  private isRunning: boolean = false;
  private rafId: number = 0;
  
  start(callback: (deltaTime: number) => void): void {
    this.isRunning = true;
    this.lastTime = performance.now() / 1000;
    
    const loop = (currentTime: number) => {
      if (!this.isRunning) return;
      
      currentTime /= 1000; // Convert to seconds
      let deltaTime = currentTime - this.lastTime;
      this.lastTime = currentTime;
      
      // Cap frame time to prevent spiral of death
      if (deltaTime > this.MAX_FRAME_TIME) {
        deltaTime = this.MAX_FRAME_TIME;
      }
      
      this.accumulator += deltaTime;
      
      // Fixed update loop
      while (this.accumulator >= this.FIXED_TIMESTEP) {
        this.fixedUpdate(this.FIXED_TIMESTEP);
        this.accumulator -= this.FIXED_TIMESTEP;
      }
      
      // Variable update
      this.update(deltaTime);
      
      // Render with interpolation alpha
      const alpha = this.accumulator / this.FIXED_TIMESTEP;
      this.render(alpha);
      
      this.rafId = requestAnimationFrame(loop);
    };
    
    this.rafId = requestAnimationFrame(loop);
  }
  
  stop(): void {
    this.isRunning = false;
    if (this.rafId) {
      cancelAnimationFrame(this.rafId);
    }
  }
}

Loop Phases

The game loop is divided into distinct phases, each handled by the LoopPhaseManager:

1. Early Update Phase

Processes input and prepares for the frame

class EarlyUpdatePhase implements LoopPhase {
  execute(deltaTime: number): void {
    // Process input
    inputManager.update(deltaTime);
    
    // Handle events
    eventEmitter.processQueue();
    
    // Update UI state
    uiStore.update();
  }
}

2. Fixed Update Phase

Physics simulation at fixed 60 Hz

class FixedUpdatePhase implements LoopPhase {
  execute(deltaTime: number): void {
    // Update physics
    physicsWorld.step(deltaTime);
    
    // Update character controllers
    characterControllers.forEach(c => c.fixedUpdate(deltaTime));
    
    // Resolve collisions
    collisionSystem.resolve();
  }
}

3. Update Phase

Game logic at variable framerate

class UpdatePhase implements LoopPhase {
  execute(deltaTime: number): void {
    // Update all systems
    systemManager.update(deltaTime);
    
    // Update entities
    entityManager.update(deltaTime);
    
    // Update scene graph
    sceneManager.update(deltaTime);
  }
}

4. Render Phase

Drawing and post-processing

class RenderPhase implements LoopPhase {
  execute(alpha: number): void {
    // Interpolate transforms for smooth rendering
    transformSystem.interpolate(alpha);
    
    // Update camera
    cameraSystem.update();
    
    // Render scene
    renderer.render(scene, camera);
    
    // Render UI
    uiRenderer.render();
  }
}

Entity Component System (ECS)

The ECS architecture separates data (Components) from logic (Systems) with Entities as lightweight identifiers.

Entity

src/engine/core/Entity.ts
class Entity {
  private id: string;
  private components: Map;
  private active: boolean = true;
  
  constructor(id: string) {
    this.id = id;
    this.components = new Map();
  }
  
  addComponent(component: Component): void {
    const type = component.constructor.name;
    this.components.set(type, component);
    component.entity = this;
  }
  
  getComponent(type: new () => T): T | null {
    return this.components.get(type.name) as T || null;
  }
  
  hasComponent(type: new () => Component): boolean {
    return this.components.has(type.name);
  }
  
  removeComponent(type: new () => Component): void {
    this.components.delete(type.name);
  }
  
  getId(): string {
    return this.id;
  }
  
  isActive(): boolean {
    return this.active;
  }
  
  setActive(active: boolean): void {
    this.active = active;
  }
}

Component

src/engine/core/component/Component.ts
abstract class Component {
  entity: Entity | null = null;
  enabled: boolean = true;
  
  // Called when component is added to entity
  onAdd(): void {}
  
  // Called when component is removed from entity
  onRemove(): void {}
  
  // Called when entity is activated
  onEnable(): void {}
  
  // Called when entity is deactivated
  onDisable(): void {}
}

// Example: Transform Component
class TransformComponent extends Component {
  position: Vector3;
  rotation: Quaternion;
  scale: Vector3;
  
  constructor() {
    super();
    this.position = new Vector3(0, 0, 0);
    this.rotation = new Quaternion();
    this.scale = new Vector3(1, 1, 1);
  }
  
  translate(offset: Vector3): void {
    this.position.add(offset);
  }
  
  rotate(rotation: Quaternion): void {
    this.rotation.multiply(rotation);
  }
}

System

src/engine/core/system/System.ts
abstract class System {
  protected world: ECSWorld;
  protected priority: number = 0;
  protected enabled: boolean = true;
  
  constructor(world: ECSWorld) {
    this.world = world;
  }
  
  // Called once when system is added
  initialize(): void {}
  
  // Called every frame
  abstract update(deltaTime: number): void;
  
  // Called when system is removed
  shutdown(): void {}
  
  // Query entities with specific components
  protected query(componentTypes: string[]): Entity[] {
    return this.world.query(componentTypes);
  }
  
  getPriority(): number {
    return this.priority;
  }
  
  isEnabled(): boolean {
    return this.enabled;
  }
}

// Example: Transform System
class TransformSystem extends System {
  update(deltaTime: number): void {
    const entities = this.query(['Transform', 'Velocity']);
    
    for (const entity of entities) {
      const transform = entity.getComponent(TransformComponent);
      const velocity = entity.getComponent(VelocityComponent);
      
      if (transform && velocity) {
        transform.translate(
          velocity.value.clone().multiplyScalar(deltaTime)
        );
      }
    }
  }
}

ECS World

src/engine/core/ECSWorld.ts
class ECSWorld {
  private entities: Map;
  private systems: System[];
  private componentIndex: Map>;
  
  constructor() {
    this.entities = new Map();
    this.systems = [];
    this.componentIndex = new Map();
  }
  
  createEntity(): Entity {
    const id = this.generateId();
    const entity = new Entity(id);
    this.entities.set(id, entity);
    return entity;
  }
  
  destroyEntity(id: string): void {
    const entity = this.entities.get(id);
    if (entity) {
      // Remove from component index
      for (const component of entity.components.values()) {
        this.removeFromIndex(entity.getId(), component);
      }
      this.entities.delete(id);
    }
  }
  
  addSystem(system: System): void {
    this.systems.push(system);
    this.systems.sort((a, b) => a.getPriority() - b.getPriority());
    system.initialize();
  }
  
  query(componentTypes: string[]): Entity[] {
    // Find entities that have ALL specified components
    const results: Entity[] = [];
    
    for (const entity of this.entities.values()) {
      if (!entity.isActive()) continue;
      
      const hasAll = componentTypes.every(type =>
        entity.hasComponent(type as any)
      );
      
      if (hasAll) {
        results.push(entity);
      }
    }
    
    return results;
  }
  
  update(deltaTime: number): void {
    for (const system of this.systems) {
      if (system.isEnabled()) {
        system.update(deltaTime);
      }
    }
  }
  
  private generateId(): string {
    return `entity_${Date.now()}_${Math.random()}`;
  }
}

Entity Management

EntityManager

Manages entity lifecycle and provides high-level entity operations:

class EntityManager {
  private world: ECSWorld;
  private factory: EntityFactory;
  private pool: EntityPool;
  
  createEntity(template?: EntityTemplate): Entity {
    const entity = this.pool.acquire();
    
    if (template) {
      this.factory.applyTemplate(entity, template);
    }
    
    return entity;
  }
  
  destroyEntity(entity: Entity): void {
    // Remove all components
    entity.removeAllComponents();
    
    // Return to pool
    this.pool.release(entity);
  }
  
  findEntitiesByTag(tag: string): Entity[] {
    return this.world.query(['Tag']).filter(e => {
      const tagComp = e.getComponent(TagComponent);
      return tagComp?.hasTag(tag);
    });
  }
  
  findEntityById(id: string): Entity | null {
    return this.world.getEntity(id);
  }
}

EntityFactory

Creates entities from templates:

class EntityFactory {
  private templates: Map;
  
  registerTemplate(name: string, template: EntityTemplate): void {
    this.templates.set(name, template);
  }
  
  create(templateName: string): Entity {
    const template = this.templates.get(templateName);
    if (!template) {
      throw new Error(`Template not found: ${templateName}`);
    }
    
    const entity = new Entity();
    this.applyTemplate(entity, template);
    return entity;
  }
  
  applyTemplate(entity: Entity, template: EntityTemplate): void {
    // Add components from template
    for (const componentData of template.components) {
      const component = this.createComponent(componentData);
      entity.addComponent(component);
    }
  }
}

EntityPool

Object pooling for entities to reduce garbage collection:

class EntityPool {
  private pool: Entity[] = [];
  private maxSize: number = 1000;
  
  acquire(): Entity {
    if (this.pool.length > 0) {
      return this.pool.pop()!;
    }
    return new Entity();
  }
  
  release(entity: Entity): void {
    if (this.pool.length < this.maxSize) {
      entity.reset();
      this.pool.push(entity);
    }
  }
  
  prewarm(count: number): void {
    for (let i = 0; i < count; i++) {
      this.pool.push(new Entity());
    }
  }
}

Scene Management

Scene

src/engine/core/Scene.ts
class Scene {
  private name: string;
  private root: SceneNode;
  private entities: Set;
  private active: boolean = false;
  
  constructor(name: string) {
    this.name = name;
    this.root = new SceneNode('root');
    this.entities = new Set();
  }
  
  addEntity(entity: Entity, parent?: SceneNode): void {
    this.entities.add(entity);
    
    const node = new SceneNode(entity.getId());
    node.entity = entity;
    
    if (parent) {
      parent.addChild(node);
    } else {
      this.root.addChild(node);
    }
  }
  
  removeEntity(entity: Entity): void {
    this.entities.delete(entity);
    // Remove from scene graph
    this.root.removeChildRecursive(entity.getId());
  }
  
  update(deltaTime: number): void {
    if (!this.active) return;
    
    // Update scene graph
    this.root.updateWorldTransforms();
  }
  
  activate(): void {
    this.active = true;
    this.onActivate();
  }
  
  deactivate(): void {
    this.active = false;
    this.onDeactivate();
  }
  
  protected onActivate(): void {}
  protected onDeactivate(): void {}
}

SceneManager

class SceneManager {
  private scenes: Map;
  private activeScene: Scene | null = null;
  
  addScene(scene: Scene): void {
    this.scenes.set(scene.getName(), scene);
  }
  
  loadScene(name: string): void {
    const scene = this.scenes.get(name);
    if (!scene) {
      throw new Error(`Scene not found: ${name}`);
    }
    
    // Deactivate current scene
    if (this.activeScene) {
      this.activeScene.deactivate();
    }
    
    // Activate new scene
    this.activeScene = scene;
    scene.activate();
  }
  
  getActiveScene(): Scene | null {
    return this.activeScene;
  }
  
  update(deltaTime: number): void {
    if (this.activeScene) {
      this.activeScene.update(deltaTime);
    }
  }
}

Scene Graph

src/engine/core/SceneNode.ts
class SceneNode {
  private id: string;
  private parent: SceneNode | null = null;
  private children: SceneNode[] = [];
  private localTransform: Transform;
  private worldTransform: Transform;
  entity: Entity | null = null;
  
  constructor(id: string) {
    this.id = id;
    this.localTransform = new Transform();
    this.worldTransform = new Transform();
  }
  
  addChild(child: SceneNode): void {
    child.parent = this;
    this.children.push(child);
  }
  
  removeChild(child: SceneNode): void {
    const index = this.children.indexOf(child);
    if (index !== -1) {
      this.children.splice(index, 1);
      child.parent = null;
    }
  }
  
  updateWorldTransforms(): void {
    // Update this node's world transform
    if (this.parent) {
      this.worldTransform = Transform.combine(
        this.parent.worldTransform,
        this.localTransform
      );
    } else {
      this.worldTransform = this.localTransform.clone();
    }
    
    // Update children recursively
    for (const child of this.children) {
      child.updateWorldTransforms();
    }
  }
  
  traverse(callback: (node: SceneNode) => void): void {
    callback(this);
    for (const child of this.children) {
      child.traverse(callback);
    }
  }
}

Performance Monitoring

PerformanceMonitor

src/engine/core/PerformanceMonitor.ts
class PerformanceMonitor {
  private fpsCounter: FPSCounter;
  private frameTimeTracker: FrameTimeTracker;
  private memoryMonitor: MemoryMonitor;
  private markers: Map;
  
  constructor() {
    this.fpsCounter = new FPSCounter();
    this.frameTimeTracker = new FrameTimeTracker();
    this.memoryMonitor = new MemoryMonitor();
    this.markers = new Map();
  }
  
  beginFrame(): void {
    this.frameTimeTracker.begin();
  }
  
  endFrame(): void {
    const frameTime = this.frameTimeTracker.end();
    this.fpsCounter.update(frameTime);
  }
  
  beginMarker(name: string): void {
    let marker = this.markers.get(name);
    if (!marker) {
      marker = new ProfilerMarker(name);
      this.markers.set(name, marker);
    }
    marker.begin();
  }
  
  endMarker(name: string): void {
    const marker = this.markers.get(name);
    if (marker) {
      marker.end();
    }
  }
  
  getStats(): PerformanceStats {
    return {
      fps: this.fpsCounter.getFPS(),
      frameTime: this.frameTimeTracker.getAverage(),
      memory: this.memoryMonitor.getUsage(),
      markers: Array.from(this.markers.values()).map(m => ({
        name: m.getName(),
        time: m.getAverage()
      }))
    };
  }
}

ModuleManager

src/engine/core/ModuleManager.ts
class ModuleManager {
  private modules: Map;
  private initOrder: string[] = [];
  
  registerModule(module: Module): void {
    const name = module.getName();
    this.modules.set(name, module);
    this.initOrder.push(name);
  }
  
  async initializeAll(): Promise {
    for (const name of this.initOrder) {
      const module = this.modules.get(name);
      if (module) {
        await module.initialize();
      }
    }
  }
  
  updateAll(deltaTime: number): void {
    for (const module of this.modules.values()) {
      module.update(deltaTime);
    }
  }
  
  shutdownAll(): void {
    // Shutdown in reverse order
    for (let i = this.initOrder.length - 1; i >= 0; i--) {
      const module = this.modules.get(this.initOrder[i]);
      if (module) {
        module.shutdown();
      }
    }
  }
  
  getModule(name: string): T | null {
    return this.modules.get(name) as T || null;
  }
}

Best Practices

ECS Design

  • Keep components as pure data containers
  • Put all logic in systems
  • Use composition over inheritance
  • Query entities efficiently using component indices

Performance

  • Use object pooling for frequently created entities
  • Profile regularly with PerformanceMonitor
  • Keep fixed timestep at 60 Hz for consistent physics
  • Batch similar operations in systems

Scene Management

  • Organize entities in a logical scene graph
  • Use scene transitions for loading/unloading
  • Clean up resources when deactivating scenes
  • Preload assets before scene activation

🎭 Animation System

Status: ✅ COMPLETE - Three.js Integration

Overview

The Animation System provides skeletal animation, blend trees, and animation state machines integrated with Three.js for smooth, professional character animation.

Architecture

Core Components

  • Skeleton - Bone hierarchy and transforms
  • AnimationClip - Keyframe data
  • AnimationMixer - Playback controller
  • AnimationAction - Individual animation
  • BlendTree - Animation blending
  • StateMachine - Animation states

Skeleton System

Bone Hierarchy

interface Bone {
  name: string;
  parent: Bone | null;
  children: Bone[];
  position: Vector3;
  rotation: Quaternion;
  scale: Vector3;
  worldMatrix: Matrix4;
  inverseBindMatrix: Matrix4;
}

class Skeleton {
  bones: Bone[];
  boneMatrices: Float32Array;
  
  update(): void;
  getBone(name: string): Bone | null;
  addBone(bone: Bone, parent?: Bone): void;
}

Usage Example

// Create skeleton
const skeleton = new Skeleton();

// Add bones
const root = skeleton.addBone({ name: 'Root' });
const spine = skeleton.addBone({ name: 'Spine', parent: root });
const head = skeleton.addBone({ name: 'Head', parent: spine });

// Update transforms
skeleton.update();

Animation Clips

Keyframe Animation

interface AnimationClip {
  name: string;
  duration: number;
  tracks: AnimationTrack[];
}

interface AnimationTrack {
  boneName: string;
  property: 'position' | 'rotation' | 'scale';
  times: number[];
  values: number[];
  interpolation: 'linear' | 'cubic' | 'step';
}

Creating Clips

const walkClip = new AnimationClip('walk', 1.0);

// Add position track
walkClip.addTrack({
  boneName: 'LeftFoot',
  property: 'position',
  times: [0, 0.5, 1.0],
  values: [
    0, 0, 0,      // Start
    0, 0.5, 0.5,  // Mid
    0, 0, 1.0     // End
  ],
  interpolation: 'cubic'
});

Animation Mixer

Playback Control

class AnimationMixer {
  play(clipName: string, options?: PlayOptions): AnimationAction;
  stop(clipName: string): void;
  pause(clipName: string): void;
  resume(clipName: string): void;
  crossFade(from: string, to: string, duration: number): void;
  update(deltaTime: number): void;
}

interface PlayOptions {
  loop?: boolean;
  speed?: number;
  weight?: number;
  fadeIn?: number;
  fadeOut?: number;
}

Usage Example

const mixer = new AnimationMixer(skeleton);

// Play animation
const action = mixer.play('walk', {
  loop: true,
  speed: 1.0,
  fadeIn: 0.3
});

// Cross-fade to run
mixer.crossFade('walk', 'run', 0.5);

// Update in game loop
mixer.update(deltaTime);

Blend Trees

Animation Blending

class BlendTree {
  addAnimation(name: string, clip: AnimationClip): void;
  setWeight(name: string, weight: number): void;
  blend(): AnimationClip;
}

// Example: Blend walk and run based on speed
const blendTree = new BlendTree();
blendTree.addAnimation('walk', walkClip);
blendTree.addAnimation('run', runClip);

// Speed = 0.5 → 50% walk, 50% run
const speed = 0.5;
blendTree.setWeight('walk', 1 - speed);
blendTree.setWeight('run', speed);

const blended = blendTree.blend();

State Machine

Animation States

class AnimationStateMachine {
  addState(name: string, clip: AnimationClip): void;
  addTransition(from: string, to: string, condition: () => boolean): void;
  setState(name: string): void;
  update(deltaTime: number): void;
}

// Example: Character movement states
const stateMachine = new AnimationStateMachine();

stateMachine.addState('idle', idleClip);
stateMachine.addState('walk', walkClip);
stateMachine.addState('run', runClip);
stateMachine.addState('jump', jumpClip);

// Transitions
stateMachine.addTransition('idle', 'walk', () => speed > 0.1);
stateMachine.addTransition('walk', 'run', () => speed > 5.0);
stateMachine.addTransition('walk', 'idle', () => speed < 0.1);
stateMachine.addTransition('*', 'jump', () => jumpPressed);

Three.js Integration

Loading GLTF Animations

import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader';

const loader = new GLTFLoader();
loader.load('/models/character.glb', (gltf) => {
  const model = gltf.scene;
  const animations = gltf.animations;
  
  // Create mixer
  const mixer = new THREE.AnimationMixer(model);
  
  // Play animations
  animations.forEach(clip => {
    const action = mixer.clipAction(clip);
    action.play();
  });
  
  // Update loop
  function animate() {
    mixer.update(deltaTime);
    renderer.render(scene, camera);
  }
});

Custom Animation Integration

// Convert our animation to Three.js format
function convertToThreeJS(clip: AnimationClip): THREE.AnimationClip {
  const tracks = clip.tracks.map(track => {
    return new THREE.VectorKeyframeTrack(
      `${track.boneName}.${track.property}`,
      track.times,
      track.values
    );
  });
  
  return new THREE.AnimationClip(clip.name, clip.duration, tracks);
}

Animation Layers

Multiple animations on different body parts:

const mixer = new AnimationMixer(skeleton);

// Layer 0: Full body walk
mixer.play('walk', { layer: 0, weight: 1.0 });

// Layer 1: Upper body wave (additive)
mixer.play('wave', { layer: 1, weight: 0.5, additive: true });

Root Motion

Extract movement from animation:

class RootMotionExtractor {

}

const motion = extractor.extract(walkClip);
character.position.add(motion.position);
character.rotation.multiply(motion.rotation);

Performance Optimization

Animation LOD

// Reduce animation quality based on distance
if (distance > 50) {
  mixer.setUpdateRate(15); // 15 FPS
} else if (distance > 20) {
  mixer.setUpdateRate(30); // 30 FPS
} else {
  mixer.setUpdateRate(60); // Full 60 FPS
}

Bone LOD

// Disable distant bones
skeleton.setBoneLOD(distance > 30 ? 'low' : 'high');

// Low LOD: Only animate root, spine, head
// High LOD: Animate all bones including fingers

Best Practices

✅ DO:

  • Use animation compression for large clips
  • Implement LOD for distant characters
  • Cache blend tree results
  • Use additive animations for layering
  • Normalize animation speeds

❌ DON'T:

  • Update all animations every frame
  • Create new mixers frequently
  • Blend more than 4 animations
  • Forget to dispose of unused clips

Examples

Character Controller with Animation

class AnimatedCharacter {
  skeleton: Skeleton;
  mixer: AnimationMixer;
  stateMachine: AnimationStateMachine;
  
  constructor() {
    this.skeleton = new Skeleton();
    this.mixer = new AnimationMixer(this.skeleton);
    this.stateMachine = new AnimationStateMachine();
    
    // Setup states
    this.setupAnimations();
  }
  
  setupAnimations() {
    this.stateMachine.addState('idle', idleClip);
    this.stateMachine.addState('walk', walkClip);
    this.stateMachine.addState('run', runClip);
    
    // Transitions
    this.stateMachine.addTransition('idle', 'walk', 
      () => this.velocity.length() > 0.1
    );
  }
  
  update(deltaTime: number) {
    // Update state machine
    this.stateMachine.update(deltaTime);
    
    // Update mixer
    this.mixer.update(deltaTime);
    
    // Update skeleton
    this.skeleton.update();
  }
}

🎨 Rendering Engine

Three.js-powered rendering with materials, shaders, lighting, and GPU instancing

Overview

The rendering engine is built on Three.js with React Three Fiber, providing a powerful and flexible rendering pipeline.

Key Features

  • ✅ React Three Fiber integration
  • ✅ Material system (Standard, PBR, Unlit, Custom)
  • ✅ Shader system with GLSL support
  • ✅ Comprehensive lighting (Ambient, Directional, Point, Spot, Area)
  • ✅ Shadow mapping
  • ✅ GPU instancing (40,000+ objects)
  • ✅ LOD (Level of Detail) system
  • ✅ Frustum culling
  • ⏳ Post-processing effects
  • ⏳ Particle systems

Material System

Material Base Class

src/engine/rendering/materials/Material.ts
abstract class Material {
  protected threeMaterial: THREE.Material;
  protected uniforms: Map;
  protected properties: MaterialProperties;
  
  constructor(properties: MaterialProperties) {
    this.properties = properties;
    this.uniforms = new Map();
  }
  
  abstract createMaterial(): THREE.Material;
  
  setUniform(name: string, value: any): void {
    this.uniforms.set(name, value);
    if (this.threeMaterial) {
      (this.threeMaterial as any).uniforms[name].value = value;
    }
  }
  
  getUniform(name: string): any {
    return this.uniforms.get(name);
  }
  
  getMaterial(): THREE.Material {
    if (!this.threeMaterial) {
      this.threeMaterial = this.createMaterial();
    }
    return this.threeMaterial;
  }
  
  dispose(): void {
    this.threeMaterial?.dispose();
  }
}

CustomMaterial

Create materials with custom shaders:

class CustomMaterial extends Material {
  private vertexShader: string;
  private fragmentShader: string;
  
  constructor(vertexShader: string, fragmentShader: string, uniforms: any) {
    super({});
    this.vertexShader = vertexShader;
    this.fragmentShader = fragmentShader;
    this.uniforms = new Map(Object.entries(uniforms));
  }
  
  createMaterial(): THREE.Material {
    return new THREE.ShaderMaterial({
      vertexShader: this.vertexShader,
      fragmentShader: this.fragmentShader,
      uniforms: Object.fromEntries(this.uniforms)
    });
  }
}

Shader System

Shader Class

src/engine/rendering/shaders/Shader.ts
class Shader {
  private name: string;
  private vertexSource: string;
  private fragmentSource: string;
  private uniforms: Map;
  private compiled: boolean = false;
  
  constructor(name: string, vertex: string, fragment: string) {
    this.name = name;
    this.vertexSource = vertex;
    this.fragmentSource = fragment;
    this.uniforms = new Map();
  }
  
  addUniform(name: string, type: string, value: any): void {
    this.uniforms.set(name, { type, value });
  }
  
  compile(): THREE.ShaderMaterial {
    const material = new THREE.ShaderMaterial({
      vertexShader: this.vertexSource,
      fragmentShader: this.fragmentSource,
      uniforms: this.getUniformsObject()
    });
    
    this.compiled = true;
    return material;
  }
  
  private getUniformsObject(): any {
    const obj: any = {};
    for (const [name, uniform] of this.uniforms) {
      obj[name] = { value: uniform.value };
    }
    return obj;
  }
}

export const standardVertexShader = `
  varying vec3 vNormal;
  varying vec3 vPosition;
  varying vec2 vUv;
  
  void main() {
    vNormal = normalize(normalMatrix * normal);
    vPosition = (modelViewMatrix * vec4(position, 1.0)).xyz;
    vUv = uv;
    
    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
  }
`;

export const standardFragmentShader = `
  uniform vec3 color;
  uniform float metalness;
  uniform float roughness;
  
  varying vec3 vNormal;
  varying vec3 vPosition;
  varying vec2 vUv;
  
  void main() {
    vec3 normal = normalize(vNormal);
    vec3 viewDir = normalize(-vPosition);
    
    // Simple lighting calculation
    float diffuse = max(dot(normal, vec3(0.0, 1.0, 0.0)), 0.0);
    vec3 finalColor = color * (0.3 + 0.7 * diffuse);
    
    gl_FragColor = vec4(finalColor, 1.0);
  }
`;

ShaderManager

class ShaderManager {
  private shaders: Map;
  private cache: ShaderCache;
  
  registerShader(shader: Shader): void {
    this.shaders.set(shader.getName(), shader);
  }
  
  getShader(name: string): Shader | null {
    return this.shaders.get(name) || null;
  }
  
  compileShader(name: string): THREE.ShaderMaterial {
    // Check cache first
    const cached = this.cache.get(name);
    if (cached) return cached;
    
    // Compile and cache
    const shader = this.getShader(name);
    if (!shader) throw new Error(`Shader not found: ${name}`);
    
    const material = shader.compile();
    this.cache.set(name, material);
    return material;
  }
}

LightingSystem

Manages all lights in the scene:

class LightingSystem {
  private lights: Map;
  private shadowMaps: Map;
  
  addLight(name: string, light: Light): void {
    this.lights.set(name, light);
    
    if (light.castsShadow()) {
      const shadowMap = new ShadowMap(light);
      this.shadowMaps.set(name, shadowMap);
    }
  }
  
  removeLight(name: string): void {
    const light = this.lights.get(name);
    if (light) {
      light.dispose();
      this.lights.delete(name);
      this.shadowMaps.delete(name);
    }
  }
  
  update(deltaTime: number): void {
    for (const light of this.lights.values()) {
      light.update(deltaTime);
    }
  }
  
  updateShadows(camera: THREE.Camera): void {
    for (const shadowMap of this.shadowMaps.values()) {
      shadowMap.update(camera);
    }
  }
}

GPU Instancing

Render thousands of objects with a single draw call using GPU instancing.

InstancedRenderer

src/engine/rendering/instancing/InstancedRenderer.ts
class InstancedRenderer {
  protected geometry: THREE.BufferGeometry;
  protected material: THREE.Material;
  protected mesh: THREE.InstancedMesh;
  protected count: number;
  
  constructor(
    geometry: THREE.BufferGeometry,
    material: THREE.Material,
    count: number
  ) {
    this.geometry = geometry;
    this.material = material;
    this.count = count;
    
    this.mesh = new THREE.InstancedMesh(
      geometry,
      material,
      count
    );
    
    this.mesh.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
  }
  
  setInstanceTransform(index: number, transform: THREE.Matrix4): void {
    this.mesh.setMatrixAt(index, transform);
    this.mesh.instanceMatrix.needsUpdate = true;
  }
  
  setInstanceColor(index: number, color: THREE.Color): void {
    if (!this.mesh.instanceColor) {
      this.mesh.instanceColor = new THREE.InstancedBufferAttribute(
        new Float32Array(this.count * 3),
        3
      );
    }
    
    this.mesh.instanceColor.setXYZ(index, color.r, color.g, color.b);
    this.mesh.instanceColor.needsUpdate = true;
  }
  
  getMesh(): THREE.InstancedMesh {
    return this.mesh;
  }
  
  dispose(): void {
    this.geometry.dispose();
    this.material.dispose();
  }
}

class GrassRenderer extends InstancedRenderer {
  private config: GrassConfig;
  private windTime: number = 0;
  
  constructor(config: GrassConfig) {
    const geometry = GrassRenderer.createBladeGeometry(
      config.bladeHeight,
      config.bladeWidth
    );
    
    const material = new THREE.ShaderMaterial({
      vertexShader: grassVertexShader,
      fragmentShader: grassFragmentShader,
      uniforms: {
        time: { value: 0 },
        windStrength: { value: config.windStrength }
      },
      side: THREE.DoubleSide
    });
    
    const count = config.density * config.areaSize * config.areaSize;
    super(geometry, material, count);
    
    this.config = config;
  }
  
  generateGrassField(getTerrainHeight: (x: number, z: number) => number): void {
    const matrix = new THREE.Matrix4();
    const position = new THREE.Vector3();
    const rotation = new THREE.Euler();
    const scale = new THREE.Vector3();
    
    let index = 0;
    const halfSize = this.config.areaSize / 2;
    const spacing = 1 / Math.sqrt(this.config.density);
    
    for (let x = -halfSize; x < halfSize; x += spacing) {
      for (let z = -halfSize; z < halfSize; z += spacing) {
        // Random offset
        const offsetX = (Math.random() - 0.5) * spacing;
        const offsetZ = (Math.random() - 0.5) * spacing;
        
        position.set(x + offsetX, 0, z + offsetZ);
        
        // Get terrain height
        position.y = getTerrainHeight(position.x, position.z);
        
        // Random rotation
        rotation.set(0, Math.random() * Math.PI * 2, 0);
        
        // Random scale variation
        const scaleVariation = 0.8 + Math.random() * 0.4;
        scale.set(scaleVariation, scaleVariation, scaleVariation);
        
        // Set transform
        matrix.compose(position, new THREE.Quaternion().setFromEuler(rotation), scale);
        this.setInstanceTransform(index, matrix);
        
        // Random color variation
        const colorVariation = 0.85 + Math.random() * 0.15;
        const color = new THREE.Color(0x4a7c59).multiplyScalar(colorVariation);
        this.setInstanceColor(index, color);
        
        index++;
      }
    }
  }
  
  update(deltaTime: number): void {
    this.windTime += deltaTime;
    (this.material as THREE.ShaderMaterial).uniforms.time.value = this.windTime;
  }

  private static createBladeGeometry(height: number, width: number): THREE.BufferGeometry {
    const geometry = new THREE.PlaneGeometry(width, height, 1, 3);
    
    // Bend the blade
    const positions = geometry.attributes.position;
    for (let i = 0; i < positions.count; i++) {

Shadow System

ShadowMap

src/engine/rendering/lighting/shadows/ShadowMap.ts
class ShadowMap {
  private light: THREE.Light;
  private resolution: number;
  private bias: number;
  private radius: number;
  
  constructor(light: THREE.Light, config?: ShadowConfig) {
    this.light = light;
    this.resolution = config?.resolution || 2048;
    this.bias = config?.bias || -0.0001;
    this.radius = config?.radius || 1;
    
    this.setupShadow();
  }
  
  private setupShadow(): void {
    this.light.castShadow = true;
    
    if (this.light.shadow) {
      this.light.shadow.mapSize.width = this.resolution;
      this.light.shadow.mapSize.height = this.resolution;
      this.light.shadow.bias = this.bias;
      this.light.shadow.radius = this.radius;
      
      // Configure shadow camera
      if (this.light instanceof THREE.DirectionalLight) {
        this.light.shadow.camera.left = -50;
        this.light.shadow.camera.right = 50;
        this.light.shadow.camera.top = 50;
        this.light.shadow.camera.bottom = -50;
        this.light.shadow.camera.near = 0.5;
        this.light.shadow.camera.far = 500;
      }
    }
  }
  
  update(camera: THREE.Camera): void {
    // Update shadow camera to follow main camera
    if (this.light.shadow && this.light instanceof THREE.DirectionalLight) {
      this.light.shadow.camera.position.copy(camera.position);
      this.light.shadow.camera.updateProjectionMatrix();
    }
  }
}

Camera System

Camera Class

src/engine/rendering/camera/Camera.ts
class Camera {
  private threeCamera: THREE.PerspectiveCamera;
  private target: Vector3;
  private offset: Vector3;
  
  constructor(fov: number = 75, aspect: number = 16/9) {
    this.threeCamera = new THREE.PerspectiveCamera(fov, aspect, 0.1, 1000);
    this.target = new Vector3();
    this.offset = new Vector3(0, 2, 5);
  }
  
  setTarget(target: Vector3): void {
    this.target.copy(target);
  }
  
  setOffset(offset: Vector3): void {
    this.offset.copy(offset);
  }
  
  update(): void {
    // Position camera relative to target
    const position = this.target.clone().add(this.offset);
    this.threeCamera.position.copy(position);
    this.threeCamera.lookAt(this.target);
  }
  
  getCamera(): THREE.PerspectiveCamera {
    return this.threeCamera;
  }
}

Level of Detail (LOD)

LODManager

src/engine/core/scene/culling/LODManager.ts
class LODManager {
  private lodLevels: Map;
  
  addLOD(entity: Entity, levels: LODLevel[]): void {
    this.lodLevels.set(entity, levels);
  }
  
  update(cameraPosition: Vector3): void {
    for (const [entity, levels] of this.lodLevels) {
      const transform = entity.getComponent(TransformComponent);
      if (!transform) continue;
      
      const distance = cameraPosition.distanceTo(transform.position);
      
      // Find appropriate LOD level
      for (let i = 0; i < levels.length; i++) {
        const level = levels[i];
        
        if (distance >= level.minDistance && distance < level.maxDistance) {
          this.setActiveLOD(entity, i);
          break;
        }
      }
    }
  }
  
  private setActiveLOD(entity: Entity, levelIndex: number): void {
    const levels = this.lodLevels.get(entity);
    if (!levels) return;
    
    // Hide all LODs
    levels.forEach(level => level.mesh.visible = false);
    
    // Show active LOD
    levels[levelIndex].mesh.visible = true;
  }
}
